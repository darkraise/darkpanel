//go:build harness

// Package vm boots QEMU/KVM virtual machines from pinned cloud images for the
// integration harness (ADR 0008, spec section 18) and drives them over SSH.
// Its methods are the fault-injection building blocks later milestones'
// tests use.
package vm

import (
	"context"
	"crypto/sha256"
	"encoding/hex"
	"fmt"
	"io"
	"net/http"
	"os"
	"path/filepath"
)

// Fetch returns the path of the image at url in cacheDir, downloading it
// unless the cache already holds a copy whose SHA-256 is sum. A download is
// checked before it enters the cache, so the cache holds verified images only.
func Fetch(ctx context.Context, url, sum, cacheDir string) (string, error) {
	if err := os.MkdirAll(cacheDir, 0o700); err != nil {
		return "", fmt.Errorf("vm: image cache: %w", err)
	}
	path := filepath.Join(cacheDir, sum+".img")
	if got, err := fileSHA256(path); err == nil && got == sum {
		return path, nil
	}
	tmp, err := os.CreateTemp(cacheDir, ".download-*")
	if err != nil {
		return "", fmt.Errorf("vm: image cache: %w", err)
	}
	defer func() { _ = os.Remove(tmp.Name()) }() // a no-op after the rename
	got, err := download(ctx, url, tmp)
	if cerr := tmp.Close(); err == nil {
		err = cerr
	}
	if err != nil {
		return "", fmt.Errorf("vm: download %s: %w", url, err)
	}
	if got != sum {
		return "", fmt.Errorf("vm: %s has SHA-256 %s, want %s; update the harness pins", url, got, sum)
	}
	if err := os.Rename(tmp.Name(), path); err != nil {
		return "", fmt.Errorf("vm: image cache: %w", err)
	}
	return path, nil
}

// download writes the body at url to w and returns its SHA-256.
func download(ctx context.Context, url string, w io.Writer) (string, error) {
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
	if err != nil {
		return "", err
	}
	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return "", err
	}
	defer func() { _ = resp.Body.Close() }()
	if resp.StatusCode != http.StatusOK {
		return "", fmt.Errorf("HTTP %s", resp.Status)
	}
	h := sha256.New()
	if _, err := io.Copy(io.MultiWriter(w, h), resp.Body); err != nil {
		return "", err
	}
	return hex.EncodeToString(h.Sum(nil)), nil
}

// fileSHA256 returns the SHA-256 of the file at path.
func fileSHA256(path string) (string, error) {
	f, err := os.Open(path) //nolint:gosec // G304: a file in the harness's own image cache
	if err != nil {
		return "", err
	}
	defer func() { _ = f.Close() }()
	h := sha256.New()
	if _, err := io.Copy(h, f); err != nil {
		return "", err
	}
	return hex.EncodeToString(h.Sum(nil)), nil
}
```

// pinned operating system, a target VM with UFW, Docker and tailscaled, and a

// PinsFile is the pin file, relative to this package's directory, which is

const PinsFile = "images.json"

// Pins are the pinned images and packages of every operating system the
// harness tests.
type Pins struct {
	Scanner string `json:"scanner"` // the OS id whose image the scanner VM boots
	OS      []OS   `json:"os"`
}

// OS is one pinned operating system.
type OS struct {
	ID         string `json:"id"`         // the -os value, such as ubuntu-22.04
	Family     string `json:"family"`     // ubuntu or debian: os-release ID and the repository path
	VersionID  string `json:"versionId"`  // os-release VERSION_ID
	Codename   string `json:"codename"`   // the apt suite
	Image      Image  `json:"image"`      // the cloud image
	Docker     string `json:"docker"`     // docker-ce and docker-ce-cli version
	Containerd string `json:"containerd"` // containerd.io version
	Tailscale  string `json:"tailscale"`  // tailscale version
}

// Image is a cloud image by dated release URL and SHA-256.
type Image struct {
	URL    string `json:"url"`
	SHA256 string `json:"sha256"`
}

var (
	sha256Hex    = regexp.MustCompile(`^[0-9a-f]{64}$`)
	datedRelease = regexp.MustCompile(`/(release-)?[0-9]{8}(-[0-9]+)?/`)
	aptName      = regexp.MustCompile(`^[a-z0-9]+$`)
	aptVersion   = regexp.MustCompile(`^[0-9A-Za-z.:~+-]+$`)
)

// LoadPins reads and checks the pin file at path.
func LoadPins(path string) (Pins, error) {
	b, err := os.ReadFile(path) //nolint:gosec // G304: the harness's own pin file
	if err != nil {
		return Pins{}, fmt.Errorf("harness: %w", err)
	}
	var p Pins
	dec := json.NewDecoder(bytes.NewReader(b))
	dec.DisallowUnknownFields()
	if err := dec.Decode(&p); err != nil {
		return Pins{}, fmt.Errorf("harness: %s: %w", path, err)
	}
	if err := p.check(); err != nil {
		return Pins{}, fmt.Errorf("harness: %s: %w", path, err)
	}
	return p, nil
}

// Find returns the pinned OS with id.
func (p Pins) Find(id string) (OS, error) {
	ids := make([]string, 0, len(p.OS))
	for _, o := range p.OS {
		if o.ID == id {
			return o, nil
		}
		ids = append(ids, o.ID)
	}
	return OS{}, fmt.Errorf("harness: unknown OS %q; the pin file lists %s", id, strings.Join(ids, ", "))
}

// check refuses duplicate ids, a scanner that names no listed OS, and any
// entry OS.check refuses.
func (p Pins) check() error {
	seen := make(map[string]bool)
	for _, o := range p.OS {
		if err := o.check(); err != nil {
			return err
		}
		if seen[o.ID] {
			return fmt.Errorf("OS %s is listed twice", o.ID)
		}
		seen[o.ID] = true
	}
	if !seen[p.Scanner] {
		return fmt.Errorf("scanner %q names no listed OS", p.Scanner)
	}
	return nil
}

// check refuses an undated or unverifiable image, and names or versions with
// characters apt does not use: the provisioning script pastes them into
// shell commands.
func (o OS) check() error {
	switch {
	case o.ID == "" || o.VersionID == "":
		return fmt.Errorf("OS %q: id and versionId are required", o.ID)
	case o.Family != "ubuntu" && o.Family != "debian":
		return fmt.Errorf("OS %s: family %q, want ubuntu or debian", o.ID, o.Family)
	case !aptName.MatchString(o.Codename):
		return fmt.Errorf("OS %s: codename %q is not a lowercase apt suite name", o.ID, o.Codename)
	case !strings.HasPrefix(o.Image.URL, "https://"):
		return fmt.Errorf("OS %s: image URL %q is not https", o.ID, o.Image.URL)
	case !datedRelease.MatchString(o.Image.URL):
		return fmt.Errorf("OS %s: image URL %q is not a dated release", o.ID, o.Image.URL)
	case !sha256Hex.MatchString(o.Image.SHA256):
		return fmt.Errorf("OS %s: image SHA-256 %q is not 64 lowercase hex digits", o.ID, o.Image.SHA256)
	}
	for _, v := range []struct{ name, version string }{
		{"docker", o.Docker}, {"containerd", o.Containerd}, {"tailscale", o.Tailscale},
	} {
		if !aptVersion.MatchString(v.version) {
			return fmt.Errorf("OS %s: %s version %q is not an apt version", o.ID, v.name, v.version)
		}
	}
	return nil
}

//go:build harness

package harness

import (
	"fmt"
	"strings"
)

// OpenPorts returns the open ports nmap's grepable output (-oG -) reports
// for host, as "<port>/<protocol>" in nmap's order. It is an error when the
// output has no port report for host, so a scan that never reached the host
// cannot pass as one that found nothing open.
func OpenPorts(grepable, host string) ([]string, error) {
	open := []string{}
	reported := false
	for _, line := range strings.Split(grepable, "\n") {
		rest, ok := strings.CutPrefix(line, "Host: ")
		if !ok {
			continue
		}
		addr, _, _ := strings.Cut(rest, " ")
		if addr != host {
			continue
		}
		for _, field := range strings.Split(rest, "\t")[1:] {
			switch {
			case strings.HasPrefix(field, "Ignored State: "):
				reported = true
			case strings.HasPrefix(field, "Ports: "):
				reported = true
				for _, p := range strings.Split(strings.TrimPrefix(field, "Ports: "), ", ") {
					// <port>/<state>/<protocol>/<owner>/<service>/<rpc>/<version>/
					f := strings.Split(p, "/")
					if len(f) >= 3 && f[1] == "open" {
						open = append(open, f[0]+"/"+f[2])
					}
				}
			}
		}
	}
	if !reported {
		return nil, fmt.Errorf("harness: nmap reported no ports for %s", host)
	}
	return open, nil
}

package vm

import (
	"bytes"
	"crypto/ed25519"
	"crypto/rand"
	"encoding/json"
	"encoding/pem"
	"fmt"
	"strings"

	"golang.org/x/crypto/ssh"
)

// User is the SSH user cloud-init creates on every VM, with passwordless
// sudo.
const User = "harness"

// McastGroup is the multicast group of the run's "internet" link.
const McastGroup = "230.0.0.1"

// Provisioning files on the VM. The runcmd wrapper writes the marker only
// when the script exits 0, so Boot tells a failed script from one that never
// ran.
const (
	provisionScript = "/usr/local/sbin/harness-provision"
	provisionLog    = "/var/log/harness-provision.log"
	provisionedMark = "/var/lib/harness-provisioned"
)

// Interface names network-config gives the NICs it matches by MAC address,
// because Ubuntu and Debian images name them differently.
const (
	mgmtIface = "dpmgmt"
	inetIface = "dpinet"
)

// Config describes one VM of a run.
type Config struct {
	Name      string // a short name, such as target or scanner: the hostname and the prefix of its files in Dir
	Dir       string // the run directory: overlay, seed, console and QEMU logs
	Image     string // a verified base image from Fetch
	MemoryMiB int
	CPUs      int
	// MgmtMAC is the user-mode management NIC, which has DHCP, outbound
	// access and the SSH port forward; InetMAC is the "internet" NIC on the
	// run's multicast link.
	MgmtMAC, InetMAC string
	// InetAddrs are the internet NIC's static addresses in CIDR form.
	InetAddrs []string
	// McastPort is the UDP port of the run's multicast link, the same for
	// every VM of the run.
	McastPort int
	// Provision is a bash script cloud-init runs once as root; Boot fails
	// unless it exits 0.
	Provision string
	// Key is the run's SSH client key; its public part is authorized for
	// User.
	Key ssh.Signer
}

// NewKey returns a new Ed25519 SSH key.
func NewKey() (ssh.Signer, error) {
	_, priv, err := ed25519.GenerateKey(rand.Reader)
	if err != nil {
		return nil, fmt.Errorf("vm: generate key: %w", err)
	}
	return ssh.NewSignerFromKey(priv)
}

// seed is the content of a NoCloud seed. cloud-init reads all three files as
// YAML, and JSON is YAML, so they are written with encoding/json rather than
// by indenting YAML by hand.
type seed struct {
	UserData, MetaData, NetworkConfig []byte
}

type cloudConfig struct {
	Hostname      string            `json:"hostname"`
	Users         []cloudUser       `json:"users"`
	SSHPwauth     bool              `json:"ssh_pwauth"`
	SSHDeleteKeys bool              `json:"ssh_deletekeys"`
	SSHKeys       map[string]string `json:"ssh_keys"`
	WriteFiles    []writeFile       `json:"write_files"`
	Runcmd        [][]string        `json:"runcmd"`
}

type cloudUser struct {
	Name              string   `json:"name"`
	Sudo              string   `json:"sudo"`
	Shell             string   `json:"shell"`
	LockPasswd        bool     `json:"lock_passwd"`
	SSHAuthorizedKeys []string `json:"ssh_authorized_keys"`
}

type writeFile struct {
	Path        string `json:"path"`
	Permissions string `json:"permissions"`
	Content     string `json:"content"`
}

type netConfig struct {
	Version   int                    `json:"version"`
	Ethernets map[string]netEthernet `json:"ethernets"`
}

type netEthernet struct {
	Match     netMatch `json:"match"`
	SetName   string   `json:"set-name"`
	DHCP4     bool     `json:"dhcp4"`
	DHCP6     bool     `json:"dhcp6"`
	AcceptRA  bool     `json:"accept-ra"`
	Addresses []string `json:"addresses,omitempty"`
}

type netMatch struct {
	MAC string `json:"macaddress"`
}

// seedFiles renders cfg's NoCloud files. hostKey becomes the VM's only SSH
// host key, which Boot then pins.
func seedFiles(cfg Config, hostKey ed25519.PrivateKey) (seed, error) {
	block, err := ssh.MarshalPrivateKey(hostKey, "")
	if err != nil {
		return seed{}, fmt.Errorf("vm %s: host key: %w", cfg.Name, err)
	}
	hostSigner, err := ssh.NewSignerFromKey(hostKey)
	if err != nil {
		return seed{}, fmt.Errorf("vm %s: host key: %w", cfg.Name, err)
	}
	user, err := marshal(cloudConfig{
		Hostname: cfg.Name,
		Users: []cloudUser{{
			Name: User, Sudo: "ALL=(ALL) NOPASSWD:ALL", Shell: "/bin/bash", LockPasswd: true,
			SSHAuthorizedKeys: []string{authorizedKey(cfg.Key.PublicKey())},
		}},
		SSHDeleteKeys: true,
		SSHKeys: map[string]string{
			"ed25519_private": string(pem.EncodeToMemory(block)),
			"ed25519_public":  authorizedKey(hostSigner.PublicKey()),
		},
		WriteFiles: []writeFile{{Path: provisionScript, Permissions: "0755", Content: cfg.Provision}},
		Runcmd: [][]string{{"bash", "-c",
			"if " + provisionScript + " >" + provisionLog + " 2>&1; then touch " + provisionedMark + "; fi"}},
	})
	if err != nil {
		return seed{}, err
	}
	meta, err := marshal(map[string]string{"instance-id": cfg.Name + "-harness", "local-hostname": cfg.Name})
	if err != nil {
		return seed{}, err
	}
	network, err := marshal(netConfig{Version: 2, Ethernets: map[string]netEthernet{
		mgmtIface: {Match: netMatch{MAC: cfg.MgmtMAC}, SetName: mgmtIface, DHCP4: true},
		inetIface: {Match: netMatch{MAC: cfg.InetMAC}, SetName: inetIface, Addresses: cfg.InetAddrs},
	}})
	if err != nil {
		return seed{}, err
	}
	return seed{UserData: append([]byte("#cloud-config\n"), user...), MetaData: meta, NetworkConfig: network}, nil
}

// authorizedKey is k in authorized_keys form without the trailing newline.
func authorizedKey(k ssh.PublicKey) string {
	return strings.TrimSpace(string(ssh.MarshalAuthorizedKey(k)))
}

func marshal(v any) ([]byte, error) {
	var b bytes.Buffer
	enc := json.NewEncoder(&b)
	enc.SetEscapeHTML(false)
	if err := enc.Encode(v); err != nil {
		return nil, fmt.Errorf("vm: seed: %w", err)
	}
	return b.Bytes(), nil
}

package vm

import (

// Limits of the VM life cycle. A first boot on a GitHub runner, with package
// installs, takes several minutes.
const (
	bootTimeout   = 25 * time.Minute
	rebootTimeout = 5 * time.Minute
	stopTimeout   = 15 * time.Second
	dialInterval  = 2 * time.Second
	dialTimeout   = 10 * time.Second
	diskSize      = "20G"
	tailLines     = 40 // log lines an error quotes
)

// VM is one running virtual machine. Its methods are safe for concurrent use.
type VM struct {
	cfg     Config
	sshAddr string        // 127.0.0.1:<port>, forwarded to the VM's port 22
	hostKey ssh.PublicKey // pinned: cloud-init installs it as the only host key
	cmd     *exec.Cmd
	exited  chan struct{} // closed when QEMU has exited
	waitErr error         // QEMU's exit, written before exited is closed

	mu     sync.Mutex
	client *ssh.Client // nil while disconnected
}

// Boot creates cfg's disk overlay and cloud-init seed in cfg.Dir, starts
// QEMU, and waits until SSH answers and the provisioning script succeeded.
// On an error the VM is stopped, and the error quotes the end of its console
// log, which stays in cfg.Dir.
func Boot(ctx context.Context, cfg Config) (*VM, error) {
	if err := os.MkdirAll(cfg.Dir, 0o700); err != nil {
		return nil, fmt.Errorf("vm %s: %w", cfg.Name, err)
	}
	image, err := filepath.Abs(cfg.Image)
	if err != nil {
		return nil, fmt.Errorf("vm %s: %w", cfg.Name, err)
	}
	_, hostKey, err := ed25519.GenerateKey(rand.Reader)
	if err != nil {
		return nil, fmt.Errorf("vm %s: host key: %w", cfg.Name, err)
	}
	hostSigner, err := ssh.NewSignerFromKey(hostKey)
	if err != nil {
		return nil, fmt.Errorf("vm %s: host key: %w", cfg.Name, err)
	}
	s, err := seedFiles(cfg, hostKey)
	if err != nil {
		return nil, err
	}
	seedPath, err := writeSeed(ctx, cfg, s)
	if err != nil {
		return nil, err
	}
	overlay := filepath.Join(cfg.Dir, cfg.Name+".qcow2")
	if err := run(ctx, "qemu-img", "create", "-q", "-f", "qcow2", "-F", "qcow2", "-b", image, overlay, diskSize); err != nil {
		return nil, err
	}
	port, err := freePort()
	if err != nil {
		return nil, fmt.Errorf("vm %s: %w", cfg.Name, err)
	}
	qemuLog, err := os.Create(qemuLogPath(cfg)) //nolint:gosec // G304: a file in the run directory
	if err != nil {
		return nil, fmt.Errorf("vm %s: %w", cfg.Name, err)
	}
	v := &VM{
		cfg: cfg, sshAddr: net.JoinHostPort("127.0.0.1", strconv.Itoa(port)),
		hostKey: hostSigner.PublicKey(), exited: make(chan struct{}),
	}
	// Not CommandContext: the VM outlives ctx, and Shutdown stops it.
	v.cmd = exec.Command("qemu-system-x86_64", qemuArgs(cfg, overlay, seedPath, port)...) //nolint:gosec // G204: arguments the harness builds
	v.cmd.Stdout, v.cmd.Stderr = qemuLog, qemuLog
	if err := v.cmd.Start(); err != nil {
		_ = qemuLog.Close()
		return nil, fmt.Errorf("vm %s: start QEMU: %w", cfg.Name, err)
	}
	go func() {
		v.waitErr = v.cmd.Wait()
		_ = qemuLog.Close()
		close(v.exited)
	}()
	bootCtx, cancel := context.WithTimeout(ctx, bootTimeout)
	defer cancel()
	if err := v.connect(bootCtx); err != nil {
		v.Shutdown()
		return nil, v.withConsole(err)
	}
	if err := v.waitProvisioned(bootCtx); err != nil {
		v.Shutdown()
		return nil, v.withConsole(err)
	}
	return v, nil
}

// qemuArgs are the QEMU arguments of cfg: KVM, the overlay and the seed as
// virtio disks, the serial console to a file, a user-mode management NIC
// forwarding sshPort on 127.0.0.1 to the VM's port 22, and the "internet"
// NIC on the run's multicast link, which needs no bridge, tap device or root.
func qemuArgs(cfg Config, overlay, seed string, sshPort int) []string {
	return []string{
		"-name", cfg.Name,
		"-machine", "q35,accel=kvm",
		"-cpu", "host",
		"-smp", strconv.Itoa(cfg.CPUs),
		"-m", strconv.Itoa(cfg.MemoryMiB),
		"-display", "none",
		"-monitor", "none",
		"-serial", "file:" + consolePath(cfg),
		"-drive", "file=" + overlay + ",if=virtio,format=qcow2",
		"-drive", "file=" + seed + ",if=virtio,format=raw,readonly=on",
		"-netdev", fmt.Sprintf("user,id=mgmt,hostfwd=tcp:127.0.0.1:%d-:22", sshPort),
		"-device", "virtio-net-pci,netdev=mgmt,mac=" + cfg.MgmtMAC,
		"-netdev", fmt.Sprintf("socket,id=inet,mcast=%s:%d", McastGroup, cfg.McastPort),
		"-device", "virtio-net-pci,netdev=inet,mac=" + cfg.InetMAC,
	}
}

func consolePath(cfg Config) string { return filepath.Join(cfg.Dir, cfg.Name+"-console.log") }

func qemuLogPath(cfg Config) string { return filepath.Join(cfg.Dir, cfg.Name+"-qemu.log") }

// writeSeed writes s into cfg.Dir and builds the NoCloud seed image with
// cloud-localds.
func writeSeed(ctx context.Context, cfg Config, s seed) (string, error) {
	base := filepath.Join(cfg.Dir, cfg.Name)
	userData, metaData, netCfg := base+"-user-data", base+"-meta-data", base+"-network-config"
	for path, data := range map[string][]byte{userData: s.UserData, metaData: s.MetaData, netCfg: s.NetworkConfig} {
		// 0600: user-data holds the VM's host private key.
		if err := os.WriteFile(path, data, 0o600); err != nil {
			return "", fmt.Errorf("vm %s: %w", cfg.Name, err)
		}
	}
	out := base + "-seed.img"
	if err := run(ctx, "cloud-localds", "-N", netCfg, out, userData, metaData); err != nil {
		return "", err
	}
	return out, nil
}

// run runs a local tool and quotes its output when it fails.
func run(ctx context.Context, name string, args ...string) error {
	out, err := exec.CommandContext(ctx, name, args...).CombinedOutput() //nolint:gosec // G204: the harness's own tool calls
	if err != nil {
		return fmt.Errorf("vm: %s: %w: %s", name, err, bytes.TrimSpace(out))
	}
	return nil
}

// freePort returns a TCP port on 127.0.0.1 that was free a moment ago.
func freePort() (int, error) {
	l, err := net.Listen("tcp4", "127.0.0.1:0")
	if err != nil {
		return 0, err
	}
	defer func() { _ = l.Close() }()
	return l.Addr().(*net.TCPAddr).Port, nil
}

// connect dials SSH until it answers, ctx ends or QEMU exits. Until
// cloud-init has created User and installed the pinned host key, the dial
// fails and is retried.
func (v *VM) connect(ctx context.Context) error {
	conf := &ssh.ClientConfig{
		User:              User,
		Auth:              []ssh.AuthMethod{ssh.PublicKeys(v.cfg.Key)},
		HostKeyCallback:   ssh.FixedHostKey(v.hostKey),
		HostKeyAlgorithms: []string{ssh.KeyAlgoED25519},
		Timeout:           dialTimeout,
	}
	for {
		c, err := ssh.Dial("tcp", v.sshAddr, conf)
		if err == nil {
			v.mu.Lock()
			v.client = c
			v.mu.Unlock()
			return nil
		}
		select {
		case <-ctx.Done():
			return fmt.Errorf("vm %s: SSH on %s did not answer: %w; last error: %v", v.cfg.Name, v.sshAddr, ctx.Err(), err)
		case <-v.exited:
			b, _ := os.ReadFile(qemuLogPath(v.cfg)) //nolint:gosec // G304: a file in the run directory
			return fmt.Errorf("vm %s: QEMU exited (%v): %s", v.cfg.Name, v.waitErr, lastLines(string(b), tailLines))
		case <-time.After(dialInterval):
		}
	}
}

// waitProvisioned waits for cloud-init to finish, then checks the marker the
// runcmd wrapper writes when the provisioning script exits 0. cloud-init
// status --wait also exits non-zero on recoverable warnings, so the marker,
// not its exit status, decides.
func (v *VM) waitProvisioned(ctx context.Context) error {
	if _, err := v.Run(ctx, "cloud-init status --wait >/dev/null 2>&1 || true"); err != nil {
		return err
	}
	if _, err := v.Run(ctx, "test -e "+provisionedMark); err == nil {
		return nil
	}
	tail, _ := v.Run(ctx, "sudo tail -n "+strconv.Itoa(tailLines)+" "+provisionLog)
	return fmt.Errorf("vm %s: provisioning failed; %s ends with:\n%s", v.cfg.Name, provisionLog, tail)
}

// Run runs cmd through User's shell on the VM and returns its standard
// output. A non-zero exit status is an error that quotes standard error; the
// standard output is still returned with it.
func (v *VM) Run(ctx context.Context, cmd string) (string, error) {
	return v.runWith(ctx, cmd, nil)
}

func (v *VM) runWith(ctx context.Context, cmd string, stdin io.Reader) (string, error) {
	v.mu.Lock()
	c := v.client
	v.mu.Unlock()
	if c == nil {
		return "", fmt.Errorf("vm %s: not connected", v.cfg.Name)
	}
	s, err := c.NewSession()
	if err != nil {
		return "", fmt.Errorf("vm %s: %w", v.cfg.Name, err)
	}
	defer func() { _ = s.Close() }()
	var stdout, stderr bytes.Buffer
	s.Stdout, s.Stderr, s.Stdin = &stdout, &stderr, stdin
	done := make(chan error, 1)
	go func() { done <- s.Run(cmd) }()
	select {
	case err := <-done:
		if err != nil {
			return stdout.String(), fmt.Errorf("vm %s: %s: %w: %s", v.cfg.Name, cmd, err, bytes.TrimSpace(stderr.Bytes()))
		}
		return stdout.String(), nil
	case <-ctx.Done():
		// The session goroutine may still write the buffers, so they are not read.
		_ = s.Close()
		return "", fmt.Errorf("vm %s: %s: %w", v.cfg.Name, cmd, ctx.Err())
	}
}

// CopyFile copies the local file src to dst on the VM as root, with mode.
func (v *VM) CopyFile(ctx context.Context, src, dst string, mode fs.FileMode) error {
	f, err := os.Open(src) //nolint:gosec // G304: a file the test names
	if err != nil {
		return fmt.Errorf("vm %s: %w", v.cfg.Name, err)
	}
	defer func() { _ = f.Close() }()
	q := shellQuote(dst)
	_, err = v.runWith(ctx, fmt.Sprintf("sudo tee %s >/dev/null && sudo chmod %o %s", q, uint32(mode.Perm()), q), f)
	return err
}

// Reboot reboots the VM and waits until SSH answers from the new boot.
func (v *VM) Reboot(ctx context.Context) error {
	before, err := v.Run(ctx, "cat /proc/sys/kernel/random/boot_id")
	if err != nil {
		return err
	}
	// The connection may drop before the exit status arrives, so the result
	// says nothing; a reboot that never happens fails the boot_id wait below.
	_, _ = v.Run(ctx, "sudo systemctl --no-block reboot")
	v.closeClient()
	ctx, cancel := context.WithTimeout(ctx, rebootTimeout)
	defer cancel()
	for {
		if err := v.connect(ctx); err != nil {
			return err
		}
		// SSH may answer once more before the old boot stops it.
		if after, err := v.Run(ctx, "cat /proc/sys/kernel/random/boot_id"); err == nil && after != before {
			return nil
		}
		v.closeClient()
		select {
		case <-ctx.Done():
			return fmt.Errorf("vm %s: no new boot: %w", v.cfg.Name, ctx.Err())
		case <-time.After(dialInterval):
		}
	}
}

// KillProcess sends SIGKILL to every process on the VM whose name is exactly
// name. It is an error when no process matched.
func (v *VM) KillProcess(ctx context.Context, name string) error {
	if _, err := v.Run(ctx, "sudo pkill -KILL -x "+shellQuote(name)); err != nil {
		return fmt.Errorf("vm %s: kill %s: %w", v.cfg.Name, name, err)
	}
	return nil
}

// ConsoleLog returns the VM's serial console output so far.
func (v *VM) ConsoleLog() (string, error) {
	b, err := os.ReadFile(consolePath(v.cfg)) //nolint:gosec // G304: a file in the run directory
	return string(b), err
}

// Shutdown stops QEMU and waits for it to exit. It is safe to call more than
// once; the disk is thrown away with the run directory.
func (v *VM) Shutdown() {
	v.closeClient()
	select {
	case <-v.exited:
		return
	default:
	}
	_ = v.cmd.Process.Signal(syscall.SIGTERM)
	select {
	case <-v.exited:
	case <-time.After(stopTimeout):
		_ = v.cmd.Process.Kill()
		<-v.exited
	}
}

func (v *VM) closeClient() {
	v.mu.Lock()
	c := v.client
	v.client = nil
	v.mu.Unlock()
	if c != nil {
		_ = c.Close()
	}
}

// withConsole adds the end of the console log to err.
func (v *VM) withConsole(err error) error {
	log, rerr := v.ConsoleLog()
	if rerr != nil {
		return fmt.Errorf("%w (console log: %v)", err, rerr)
	}
	return fmt.Errorf("%w\nconsole %s ends with:\n%s", err, consolePath(v.cfg), lastLines(log, tailLines))
}

// lastLines returns the last n lines of s.
func lastLines(s string, n int) string {
	lines := strings.Split(strings.TrimRight(s, "\n"), "\n")
	if len(lines) > n {
		lines = lines[len(lines)-n:]
	}
	return strings.Join(lines, "\n")
}

// shellQuote quotes s as one word for a POSIX shell.
func shellQuote(s string) string {
	return "'" + strings.ReplaceAll(s, "'", `'\''`) + "'"
}

// checkKillProcess kills the running panel and checks that it is gone.

func unitState(ctx context.Context, v *vm.VM, unit string) string {
	out, _ := v.Run(ctx, "systemctl is-active "+unit+" || true")

// eventually calls check every pollInterval until it returns nil or timeout
// passes, and then returns its last error.
func eventually(ctx context.Context, timeout time.Duration, check func() error) error {
	deadline := time.Now().Add(timeout)
	for {
		err := check()
		if err == nil || time.Now().After(deadline) {
			return err
		}
		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-time.After(pollInterval):
		}
	}
}

// planLocks are per-plan locks of this process. One process is enough: the
// controller lock keeps the controller to one instance.
type planLocks struct {
	mu   sync.Mutex
	held map[string]*planLock
}

type planLock struct {
	slot  chan struct{} // holds a value while the plan is locked
	users int           // holders and waiters; guarded by planLocks.mu
}

// lock locks planID, waiting until it is free or ctx ends, and returns the
// function that unlocks it.
func (l *planLocks) lock(ctx context.Context, planID string) (func(), error) {
	l.mu.Lock()
	if l.held == nil {
		l.held = make(map[string]*planLock)
	}
	pl := l.held[planID]
	if pl == nil {
		pl = &planLock{slot: make(chan struct{}, 1)}
		l.held[planID] = pl
	}
	pl.users++
	l.mu.Unlock()
	release := func() {
		l.mu.Lock()
		if pl.users--; pl.users == 0 {
			delete(l.held, planID)
		}
		l.mu.Unlock()
	}
	select {
	case pl.slot <- struct{}{}:
		return func() {
			<-pl.slot
			release()
		}, nil
	case <-ctx.Done():
		release()
		return nil, ctx.Err()
	}
}
```

// connectNetwork returns the client of the network API. The URL comes from
// flagURL, then DARKPANEL_URL, then the config file, then DefaultURL; the
// token from DARKPANEL_TOKEN, then the config file, whose token goes only to
// the config file's server (its URL, or DefaultURL when it names none); the
// CA fingerprint from the config file. A token is never sent over plain http
// to a host that is not loopback (spec section 15, register row 91). The
// config file is read whenever it exists.
func connectNetwork(flagURL string, getenv func(string) string, home func() (string, error)) (*client, error) {
	var cfg fileConfig
	if path, err := configPath(getenv, home); err == nil {
		if cfg, err = loadConfig(path); err != nil {
			return nil, err
		}
	}
	raw := firstOf(flagURL, getenv(EnvURL), cfg.URL, DefaultURL)
	u, err := url.Parse(raw)
	if err != nil || (u.Scheme != "http" && u.Scheme != "https") || u.Host == "" {
		return nil, usagef("server URL %q: want http://<host>:<port> or https://<host>:<port>", raw)
	}
	token := getenv(EnvToken)
	if token == "" && sameServer(u, firstOf(cfg.URL, DefaultURL)) {
		token = cfg.Token
	}
	if token != "" && strings.EqualFold(u.Scheme, "http") && !isLoopbackHost(u.Hostname()) {
		return nil, usagef("refusing to send the API token over plain http to %s; use https, or http to a loopback address", u.Host)
	}
	hc, err := newHTTPClient(u, cfg.CAFingerprint)
	if err != nil {
		return nil, err
	}
	return &client{base: strings.TrimSuffix(u.String(), "/"), token: token, http: hc}, nil
}

// sameServer reports whether u names the server of the URL raw: the same
// scheme, host and port, and the same path but for a trailing slash.
func sameServer(u *url.URL, raw string) bool {
	f, err := url.Parse(raw)
	return err == nil && strings.EqualFold(u.Scheme, f.Scheme) && strings.EqualFold(u.Host, f.Host) &&
		strings.TrimSuffix(u.Path, "/") == strings.TrimSuffix(f.Path, "/")
}

// isLoopbackHost reports whether host, a URL host without its port, is
// localhost or a loopback IP address.
func isLoopbackHost(host string) bool {
	if strings.EqualFold(host, "localhost") {
		return true
	}
	ip := net.ParseIP(host)
	return ip != nil && ip.IsLoopback()
}
